import { useState, useEffect } from 'react';

export default function TaskList({ projectId }) {
  const [tasks, setTasks] = useState([]);
  const [taskName, setTaskName] = useState('');

  useEffect(() => {
    const fetchTasks = async () => {
      const res = await fetch(`/api/projects/${projectId}/tasks`);
      const data = await res.json();
      setTasks(data);
    };
    fetchTasks();
  }, [projectId]);

  const addTask = async () => {
    const res = await fetch('/api/tasks', {
      method: 'POST',
      body: JSON.stringify({ name: taskName, projectId }),
      headers: { 'Content-Type': 'application/json' }
    });
    const newTask = await res.json();
    setTasks(prev => [...prev, newTask]);
    setTaskName('');
  };

  return (
    <div>
      <input value={taskName} onChange={e => setTaskName(e.target.value)} />
      <button onClick={addTask}>Add Task</button>
      <ul>
        {tasks.map(task => <li key={task._id}>{task.name}</li>)}
      </ul>
    </div>
  );
}
✅ Final Notes
Avoid fetching only once with useEffect([]) if your UI depends on data changes.

Use setState() immediately after a successful API call to reflect changes.

When using libraries like React Query or SWR, ensure to mutate() or invalidateQueries() appropriately.

Always make sure components rerender based on updated state or props.
